cmake_minimum_required(VERSION 3.22)
project(PitcherDesignStudy VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
set(PITCHER_JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/_deps/juce-src" CACHE PATH "Existing JUCE 8 checkout")
if(NOT EXISTS "${PITCHER_JUCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set PITCHER_JUCE_DIR to an existing JUCE checkout; this study does not download dependencies.")
endif()
add_subdirectory("${PITCHER_JUCE_DIR}" juce EXCLUDE_FROM_ALL)
add_subdirectory(Source/ui)
juce_add_binary_data(StudyAssets SOURCES Assets/study.json ../../assets/fonts/Inter-Regular.ttf ../../assets/fonts/Inter-Bold.ttf ../../assets/fonts/SpaceMono-Regular.ttf)
juce_add_gui_app(PitcherStudy PRODUCT_NAME "Pitcher Design Study" COMPANY_NAME "Pitcher" BUNDLE_ID "org.pitcher.design-study")
target_sources(PitcherStudy PRIVATE Source/Main.cpp Source/StudyView.cpp Source/StudyData.cpp Source/StudyOrbit.cpp)
target_compile_definitions(PitcherStudy PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:PitcherStudy,JUCE_PRODUCT_NAME>" JUCE_APPLICATION_VERSION_STRING="0.1.0")
target_link_libraries(PitcherStudy PRIVATE StudyAssets pitcher_study_ui juce::juce_gui_extra juce::juce_audio_utils juce::juce_recommended_config_flags)
